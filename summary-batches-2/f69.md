private fun cleanupOldLogs() {
        val files =
            logsDir.listFiles()
                ?.filter { it.isFile }
                ?.filterNot { it.name == currentFileName }
                ?.sortedByDescending { it.lastModified() }
                .orEmpty()

        val allowed = (maxLogFiles - 1).coerceAtLeast(0)
        if (files.size <= allowed) return

        files.drop(allowed).forEach { it.delete() }
    }
~~~~
